import SwiftUI
struct ContentView:View{
    @State var ShowMessage = false
    @State var name =""
    var body:some View{ 
        VStack{ 
            Text("My project")
            .foregroundColor(.red)
            Text("Enter my name")
            TextField("name",text:($name))
            Button("submit"){
                ShowMessage=true
            }
            .frame(width:80,height:40)
            .background(Color.black)
            .foregroundColor(.blue)
            .cornerRadius(20)
            if ShowMessage{
                Text("my name :\($name)")
            }
        }

    }
}
