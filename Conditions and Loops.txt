    print("You passed")
} else {
    print("You failed")
}

//learn else if
let mark = 75

if mark >= 80 {
    print("Excellent")

}

//Loops
for number in 1...5 {
    print("Hello")
}

//Print numbers from 1 to 5
for number in 1...5 {
    print(number)
}

//Closed range(1...5)
for number in 1...5 {
    print(number)
}

//Half-open range(1..<5)
for number in 1..<5 {
  print(number)
}

//multiplication table for 2.
for number in 1...5 {
    print("2 x \(number) = \(2 * number)")
}

//Count how many times the loop runs
var count = 0
for number in 1...10 {
    count += 1
}
print("Total count: \(count)")

//Count only even numbers
var count = 0
for number in 1...10 {
    if number % 2 == 0 {
        print(number)
        count += 1
    }
}
print("Total even numbers: \(count)")
